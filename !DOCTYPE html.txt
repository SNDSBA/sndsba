    <script>
        // Generate realistic localized data points from Year 5 to 22 based on MPOB production curves
        const startYear = 5;
        const endYear = 22;
        const yearsData = [];
        const yieldData = [];
        const rateData = [];

        // Cubic model: f(x) = -0.05x^3 + 1.8x^2 - 12x + 25
        function calculateYield(x) {
            return (-0.05 * Math.pow(x, 3)) + (1.8 * Math.pow(x, 2)) - (12 * x) + 25;
        }

        // Derivative: f'(x) = -0.15x^2 + 3.6x - 12
        function calculateDerivative(x) {
            return (-0.15 * Math.pow(x, 2)) + (3.6 * x) - 12;
        }

        // Populate datasets
        for (let x = startYear; x <= endYear; x++) {
            yearsData.push(x);
            yieldData.push(parseFloat(calculateYield(x).toFixed(2)));
            rateData.push(parseFloat(calculateDerivative(x).toFixed(2)));
        }

        // Initialize Chart.js
        const ctx = document.getElementById('yieldChart').getContext('2d');
        const yieldChart = new Chart(ctx, {
            type: 'line',
            data: {
                labels: yearsData,
                datasets: [{
                    label: 'FFB Yield f(x) (t/ha)',
                    data: yieldData,
                    borderColor: '#10b981', // Emerald
                    borderWidth: 4,
                    backgroundColor: 'rgba(16, 185, 129, 0.1)',
                    fill: true,
                    tension: 0.3,
                    pointBackgroundColor: '#34d399',
                    pointHoverRadius: 9,
                    pointHoverBackgroundColor: '#f59e0b' // Amber on hover
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: {
                    legend: {
                        labels: { color: '#e5e7eb', font: { weight: 'bold' } }
                    },
                    tooltip: {
                        callbacks: {
                            label: function(context) {
                                return ` Yield: ${context.parsed.y} t/ha (Click to view growth rate)`;
                            }
                        }
                    }
                },
                scales: {
                    x: {
                        grid: { color: 'rgba(31, 41, 55, 0.5)' },
                        ticks: { color: '#9ca3af' },
                        title: { display: true, text: 'Tree Age (Years Since Planting)', color: '#9ca3af' }
                    },
                    y: {
                        grid: { color: 'rgba(31, 41, 55, 0.5)' },
                        ticks: { color: '#9ca3af' },
                        title: { display: true, text: 'Fresh Fruit Bunch Yield (Tonnes/Hectare)', color: '#9ca3af' },
                        min: 5,
                        max: 35
                    }
                },
                // Trigger action when user clicks anywhere on the line/points
                onClick: (event, activeElements) => {
                    if (activeElements.length > 0) {
                        const index = activeElements[0].index;
                        const selectedYear = yearsData[index];
                        processCalculusOutput(selectedYear);
                    }
                }
            }
        });

        // Compute derivatives and generate natural-language insights dynamically
        function processCalculusOutput(year) {
            const yieldVal = calculateYield(year).toFixed(2);
            const rateVal = calculateDerivative(year).toFixed(2);
            
            // Elements mapping
            document.getElementById('placeholderText').classList.add('hidden');
            const calcResultElement = document.getElementById('calcResult');
            calcResultElement.classList.remove('hidden');
            
            document.getElementById('outYear').innerText = year;
            document.getElementById('outYield').innerText = yieldVal;
            
            const rateNode = document.getElementById('outRate');
            if (rateVal > 0) {
                rateNode.innerText = `+${rateVal}`;
                rateNode.className = "text-xl font-bold text-emerald-400 font-mono";
            } else {
                rateNode.innerText = rateVal;
                rateNode.className = "text-xl font-bold text-rose-400 font-mono";
            }

            // Generate field-specific natural text interpretations based on derivative value thresholds
            let interpretation = "";
            if (year >= 5 && year <= 8) {
                interpretation = `At Year ${year}, the derivative is positive ($f'(x) = ${rateVal}$), meaning the tree is entering its rapid structural expansion phase. Yield acceleration is high. Focus heavily on Nitrogen and Potassium fertilizers right now to capitalize on this exponential growth surge.`;
            } else if (year > 8 && year <= 11) {
                interpretation = `At Year ${year}, the growth rate is positive but beginning to level off ($f'(x) = ${rateVal}$). The tree is near its optimal peak production efficiency. Maintain consistent plantation management; this is your highest profit-yielding window.`;
            } else if (year == 12) {
                interpretation = `At Year 12, the growth rate approaches absolute flat zero ($f'(x) = ${rateVal}$). This marks the mathematical maximum ceiling of your crop yield. Production cannot naturally expand further beyond this point.`;
            } else if (year > 12 && year <= 18) {
                interpretation = `The derivative has crossed into negative territory ($f'(x) = ${rateVal}$). This means that while you still harvest a substantial total quantity, the biological capacity of the palm is degrading year-on-year. Yield is losing velocity.`;
            } else {
                interpretation = `At Year ${year}, the derivative shows steep losses ($f'(x) = ${rateVal}$). Biological senility has firmly set in. It is critical to review your economic thresholds to schedule dynamic field-replanting strategies.`;
            }

            document.getElementById('laymanExplanation').innerHTML = interpretation;
            
            // Simple visual entry effect for the display card
            const outputBlock = document.getElementById('outputNode');
            outputBlock.classList.add('scale-[1.01]', 'border-amber-500/40');
            setTimeout(() => {
                outputBlock.classList.remove('scale-[1.01]', 'border-amber-500/40');
            }, 200);
        }
    </script>
